2. **Rewrite the skill injection block** (around lines 651-666). Currently it calls `injectSkills()` to copy skill directories. Replace with JIT installation via `npx skills add`:

```typescript
// Install matching skills into project's .opencode/skill/ directory via JIT npx
const resolvedSkills = resolveSkillsForJob(job.categories ?? null);
const installedSkillNames: string[] = [];
if (resolvedSkills.length > 0) {
  for (const skill of resolvedSkills) {
    try {
      await execa('npx', [
        'skills', 'add', skill.repo,
        '--skill', skill.skill,
        '--agent', 'opencode',
        '--yes',
      ], {
        cwd: projectDir,
        timeout: 60_000,
      });
      installedSkillNames.push(skill.name);
    } catch (err) {
      process.stderr.write(
        `[runner] Warning: skill install failed for ${skill.name}: ${errMsg(err)}\n`,
      );
    }
  }
  if (installedSkillNames.length > 0) {
    process.stderr.write(
      `[runner] Installed ${installedSkillNames.length} skill(s) for job ${job.id}: ${installedSkillNames.join(', ')}\n`,
    );
  }
}
```
