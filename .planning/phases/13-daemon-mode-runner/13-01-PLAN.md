  4. **Add `findLaunchableAtomic`** — holds lock during read + mark to prevent TOCTOU:
     ```typescript
     async function findLaunchableAtomic(
       runningProjects: Set<string>,
       maxParallel: number,
       activeCount: number,
     ): Promise<QueueJsonItem | null> {
       return withQueueJsonLock(async () => {
         const data = await loadQueue();
         if (activeCount >= maxParallel) return null;

         for (const item of data.items) {
           if (item.status !== 'queued') continue;
           if (runningProjects.has(item.project)) continue;
           if (item.dependsOn !== null) {
             if (!data.completedIds.includes(item.dependsOn)) continue;
           }
           // Found launchable — mark running atomically
           item.status = 'running';
           item.startedAt = new Date().toISOString();
           item.attempts++;
           await saveQueue(data);
           return item;
         }
         return null;
       });
     }
     ```
     Export this new function. The runner will use `findLaunchableAtomic` instead of separate `findLaunchable` + `markRunning`.

  5. **Add `markBlocked`** — mark an item as blocked with error reason:
     ```typescript
     async function markBlocked(id: string, reason: string): Promise<void> {
       return withQueueJsonLock(async () => {
         const data = await loadQueue();
         const item = data.items.find((i) => i.id === id);
         if (!item) throw new Error(`Item not found: ${id}`);
         item.status = 'blocked';
         item.error = reason;
         await saveQueue(data);
       });
     }
     ```

  6. **Add `cascadeFailure`** — when a job permanently fails, block all transitive dependents:
     ```typescript
     async function cascadeFailure(failedId: string): Promise<string[]> {
       return withQueueJsonLock(async () => {
         const data = await loadQueue();
         const blockedIds: string[] = [];

         // Find all items that directly or transitively depend on failedId
         const toBlock = new Set<string>();
         const queue = [failedId];

         while (queue.length > 0) {
           const currentId = queue.shift()!;
           for (const item of data.items) {
             if (item.dependsOn === currentId && item.status === 'queued' && !toBlock.has(item.id)) {
               toBlock.add(item.id);
               queue.push(item.id);  // Check for transitive deps
             }
           }
         }

         for (const id of toBlock) {
           const item = data.items.find((i) => i.id === id);
           if (item) {
             item.status = 'blocked';
             item.error = `dependency ${failedId} failed`;
             blockedIds.push(id);
           }
         }

         if (blockedIds.length > 0) {
           await saveQueue(data);
         }
         return blockedIds;
       });
     }
     ```
     Export this function. Returns list of blocked item IDs for logging.
