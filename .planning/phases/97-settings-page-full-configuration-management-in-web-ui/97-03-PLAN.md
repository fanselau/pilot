// In server-fns.ts
export const addProviderModeFn = createServerFn({ method: 'POST' })
  .inputValidator((d: { name: string; copyFrom?: string }) => d)
  .handler(async ({ data }) => {
    const { addProviderMode, cloneProviderMode } = await import('@pilot/core/model-store.js')
    if (data.copyFrom) {
      cloneProviderMode(data.copyFrom, data.name)
    } else {
      addProviderMode(data.name)
    }
    return { ok: true }
  })

export const removeProviderModeFn = createServerFn({ method: 'POST' })
  .inputValidator((d: { name: string }) => d)
  .handler(async ({ data }) => {
    const { removeProviderMode } = await import('@pilot/core/model-store.js')
    removeProviderMode(data.name)
    return { ok: true }
  })
