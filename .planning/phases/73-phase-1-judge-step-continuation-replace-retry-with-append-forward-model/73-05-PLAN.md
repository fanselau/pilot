const steps = getJobSteps(job.id);
if (steps.length > 0) {
  const stepLines = steps.map((step, i) => {
    const statusIcon = step.status === 'completed' ? '✓' : step.status === 'failed' ? '✗' : step.status === 'pending' ? '○' : '◆';
    const sourceTag = step.source !== 'delegation' ? ` [${step.source}]` : '';
    const errorSuffix = step.error ? ` — ${step.error.slice(0, 80)}` : '';
    return `  ${statusIcon} Step ${i + 1}: ${step.command} ${step.args.slice(0, 40)}${sourceTag}${errorSuffix}`;

  // Append to notification content
  stepHistorySection = `\n\nStep history:\n${stepLines}`;
}
