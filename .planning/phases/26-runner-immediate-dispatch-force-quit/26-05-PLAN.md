describe('claimNextLaunchable', () => {
  it('returns null when no pending jobs exist', () => {
    const result = claimNextLaunchable();
    expect(result).toBeNull();
  });

  it('claims a pending job and marks it running atomically', () => {
    const job = addJob('proj-a', 'quick', 'task 1');
    const claimed = claimNextLaunchable();
    expect(claimed).not.toBeNull();
    expect(claimed!.id).toBe(job.id);
    expect(claimed!.status).toBe('running');
    // Verify DB was updated too
    const fresh = getJob(job.id)!;
    expect(fresh.status).toBe('running');
    expect(fresh.startedAt).not.toBeNull();
    expect(fresh.attempts).toBe(1);
  });

  it('respects project serialization: does not claim if project already running', () => {
    const job1 = addJob('proj-x', 'quick', 'task 1');
    const job2 = addJob('proj-x', 'quick', 'task 2');  // same project
    
    // Claim first job
    const first = claimNextLaunchable();
    expect(first!.id).toBe(job1.id);
    
    // Second job from same project should NOT be claimed
    const second = claimNextLaunchable();
    expect(second).toBeNull();
    
    // job2 remains pending
    const j2 = getJob(job2.id)!;
    expect(j2.status).toBe('pending');
  });

  it('claims from a different project while one project is running', () => {
    const jobA = addJob('proj-a', 'quick', 'task 1');
    const jobB = addJob('proj-b', 'quick', 'task 2');  // different project
    
    // Claim proj-a job
    claimNextLaunchable();
    expect(getJob(jobA.id)!.status).toBe('running');
    
    // proj-b job should be claimable
    const claimed = claimNextLaunchable();
    expect(claimed).not.toBeNull();
    expect(claimed!.id).toBe(jobB.id);
    expect(claimed!.status).toBe('running');
  });

  it('returns null when only running jobs exist (all pending claimed)', () => {
    addJob('proj-a', 'quick', 'task 1');
    claimNextLaunchable();  // claims it
    const second = claimNextLaunchable();
    expect(second).toBeNull();
  });
});

describe('immediate dispatch', () => {
  it('fills multiple slots in one iteration when capacity exists', async () => {
    // Arrange: mock claimNextLaunchable to return two jobs then null
    const job1 = makeJob('j1', 'proj-a');
    const job2 = makeJob('j2', 'proj-b');
    let claimCount = 0;
    mockClaimNextLaunchable.mockImplementation(() => {
      if (claimCount === 0) { claimCount++; return job1; }
      if (claimCount === 1) { claimCount++; return job2; }
      return null;
    });
    mockDelegate.mockResolvedValue({ steps: [] });
    
    const runner = createRunner({ maxParallel: 2, once: true, pollInterval: 60 });
    await runner.run();
    
    // Both jobs should have been launched
    expect(mockClaimNextLaunchable).toHaveBeenCalledTimes(3); // job1, job2, null
    expect(mockMarkCompleted).toHaveBeenCalledWith('j1');
    expect(mockMarkCompleted).toHaveBeenCalledWith('j2');
  });

  it('stops claiming when maxParallel reached', async () => {
    const job1 = makeJob('j1', 'proj-a');
    const job2 = makeJob('j2', 'proj-b');
    let claimCount = 0;
    mockClaimNextLaunchable.mockImplementation(() => {
      if (claimCount < 2) { claimCount++; return claimCount === 1 ? job1 : job2; }
      return null;
    });
    mockDelegate.mockResolvedValue({ steps: [] });
    
    const runner = createRunner({ maxParallel: 1, once: true, pollInterval: 60 });
    await runner.run();
    
    // With maxParallel=1, drain loop behavior:
    //   iteration 1: claim j1 (slot fills, loop exits inner drain), j1 completes
    //   iteration 2: claim j2 (slot fills, loop exits inner drain), j2 completes
    //   iteration 3: claim null (no more jobs), runner exits
    // Total: 3 calls to claimNextLaunchable
    expect(mockClaimNextLaunchable).toHaveBeenCalledTimes(3); // j1, j2 (after j1 done), null
    // Verify sequential ordering: j1 was launched before j2
    expect(mockMarkCompleted).toHaveBeenNthCalledWith(1, 'j1');
    expect(mockMarkCompleted).toHaveBeenNthCalledWith(2, 'j2');
  });
});

import { killCommand } from '../../src/commands/kill.js';

const makeRunningJob = (id = 'abc1') => ({
  id,
  project: 'my-project',
  status: 'running',
  sessionTitles: JSON.stringify([`my-project-execute-phase-${id}`]),
  scope: 'quick',
  description: 'test',
  priority: 0,
  dependsOn: null,
  createdAt: new Date().toISOString(),
  startedAt: new Date().toISOString(),
  completedAt: null,
  error: null,
  attempts: 1,

  delegationPlan: null,
  currentStep: 0,
  requirementPath: null,
  modelProfile: 'balanced',
  providerMode: 'claude-only',
});

describe('killCommand', () => {
  beforeEach(() => {
    vi.clearAllMocks();
    vi.spyOn(process, 'exit').mockImplementation((code) => { throw new Error(`exit:${code}`); });
    vi.spyOn(process.stderr, 'write').mockImplementation(() => true);
  });

  it('exits 1 when job not found', async () => {
    mockGetJob.mockReturnValue(null);
    await expect(killCommand('notfound', { force: true })).rejects.toThrow('exit:1');
    expect(process.stderr.write).toHaveBeenCalledWith(expect.stringMatching(/not found/));
  });

  it('exits 1 without --force flag', async () => {
    mockGetJob.mockReturnValue(makeRunningJob());
    await expect(killCommand('abc1', {})).rejects.toThrow('exit:1');
    expect(process.stderr.write).toHaveBeenCalledWith(expect.stringMatching(/--force/));
  });

  it('exits 1 for non-running job', async () => {
    mockGetJob.mockReturnValue({ ...makeRunningJob(), status: 'pending' });
