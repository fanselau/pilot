describe('parseIntentOutput', () => {
  // ── Valid intent types ────────────────────────────────

  it('parses quick intent from fenced JSON block', () => {
    const content = [
      'Here is the delegation result:',
      '```json',
      '{"intent":{"type":"quick","description":"Fix the bug"},"reasoning":"Simple fix"}',
      '```',
    ].join('\n');
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('quick');
    expect(result.intent).toHaveProperty('description', 'Fix the bug');
    expect(result.reasoning).toBe('Simple fix');
  });

  it('parses quick intent with flags', () => {
    const content = '```json\n{"intent":{"type":"quick","description":"Add tests","flags":["full"]},"reasoning":"Quality-critical"}\n```';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('quick');
    if (result.intent.type === 'quick') {
      expect(result.intent.flags).toEqual(['full']);
    }
  });

  it('parses init-project intent', () => {
    const content = '```json\n{"intent":{"type":"init-project","prdPath":"requirements/foo.md"},"reasoning":"No .planning/ dir"}\n```';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('init-project');
    if (result.intent.type === 'init-project') {
      expect(result.intent.prdPath).toBe('requirements/foo.md');
    }
  });

  it('parses new-milestone intent', () => {
    const content = '```json\n{"intent":{"type":"new-milestone","prdPath":"requirements/v2.md"},"reasoning":"Need new milestone"}\n```';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('new-milestone');
  });

  it('parses plan-and-execute intent', () => {
    const content = '```json\n{"intent":{"type":"plan-and-execute","phaseNumber":42,"prdPath":"requirements/foo.md","addPhaseTitle":"Feature Name"},"reasoning":"Phase 42 needs planning"}\n```';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('plan-and-execute');
    if (result.intent.type === 'plan-and-execute') {
      expect(result.intent.phaseNumber).toBe(42);
      expect(result.intent.prdPath).toBe('requirements/foo.md');
      expect(result.intent.addPhaseTitle).toBe('Feature Name');
    }
  });

  it('parses plan-and-execute intent with isGapClosure', () => {
    const content = '```json\n{"intent":{"type":"plan-and-execute","phaseNumber":5,"isGapClosure":true},"reasoning":"Gap retry"}\n```';
    const result = parseIntentOutput(content);
    if (result.intent.type === 'plan-and-execute') {
      expect(result.intent.isGapClosure).toBe(true);
    }
  });

  it('parses execute-only intent', () => {
    const content = '```json\n{"intent":{"type":"execute-only","phaseNumber":10},"reasoning":"Plans exist"}\n```';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('execute-only');
    if (result.intent.type === 'execute-only') {
      expect(result.intent.phaseNumber).toBe(10);
    }
  });

  it('parses audit-milestone intent', () => {
    const content = '```json\n{"intent":{"type":"audit-milestone","version":"v1.0"},"reasoning":"All phases complete"}\n```';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('audit-milestone');
    if (result.intent.type === 'audit-milestone') {
      expect(result.intent.version).toBe('v1.0');
    }
  });

  it('parses noop intent', () => {
    const content = '```json\n{"intent":{"type":"noop","reason":"Phase 5 already complete"},"reasoning":"Nothing to do"}\n```';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('noop');
    if (result.intent.type === 'noop') {
      expect(result.intent.reason).toBe('Phase 5 already complete');
    }
  });

  it('parses raw JSON (no fenced block)', () => {
    const content = '{"intent":{"type":"quick","description":"Fix bug"},"reasoning":"Simple"}';
    const result = parseIntentOutput(content);
    expect(result.intent.type).toBe('quick');
  });

  it('defaults reasoning to empty string when missing', () => {
    const content = '```json\n{"intent":{"type":"noop","reason":"Done"}}\n```';
    const result = parseIntentOutput(content);
    expect(result.reasoning).toBe('');
  });

  // ── Error cases ────────────────────────────────────────

  it('throws on invalid JSON', () => {
    expect(() => parseIntentOutput('not json')).toThrow('Failed to parse');
  });

  it('throws on missing intent field', () => {
    const content = '```json\n{"reasoning":"No intent"}\n```';
    expect(() => parseIntentOutput(content)).toThrow();
  });

  it('throws on unknown intent type', () => {
    const content = '```json\n{"intent":{"type":"unknown-thing"},"reasoning":"Bad"}\n```';
    expect(() => parseIntentOutput(content)).toThrow('Invalid intent type');
  });

  it('throws when intent is null', () => {
    const content = '```json\n{"intent":null,"reasoning":"Bad"}\n```';
    expect(() => parseIntentOutput(content)).toThrow();
  });
});
