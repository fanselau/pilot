       ```tsx
       <box borderStyle="rounded" borderColor={props.focused ? statusColors.running : theme.border} flexGrow={1}>
         <text content=" Queue " /> {/* title */}

           <For each={props.jobs}>
             {(job, i) => (
               <box backgroundColor={props.focused && i() === props.selectedIndex ? theme.highlight : undefined}>
                 <text
                   content={`${props.focused && i() === props.selectedIndex ? '▸' : ' '} #${job.id}  ${job.project}  ${job.scope}  "${truncate(job.description, 30)}"`}

                 />
               </box>

           </For>

       </box>
       ```
