```typescript
import { accessSync, readFileSync, statSync } from 'node:fs';
import path from 'node:path';
import { getConfig } from '../core/config.js';
import { addJob } from '../core/db.js';
import { outputJson, outputHuman, isJsonMode } from '../util/output.js';
import { green, dim, cyan } from '../util/colors.js';
import type { JobScope } from '../core/types.js';

interface AddOptions {
  as?: JobScope;
  next?: boolean;
  json?: boolean;
}

function isFilePath(str: string): boolean {
  try { accessSync(str); return statSync(str).isFile(); } catch { return false; }
}

function isDirPath(str: string): boolean {
  try { return statSync(str).isDirectory(); } catch { return false; }
}

/**
 * Detect scope from requirement type:
 * - File path → phase
 * - Directory → milestone
 * - Short string → quick
 */
function detectScope(requirement: string): JobScope {
  if (isFilePath(requirement)) return 'phase';
  if (isDirPath(requirement)) return 'milestone';
  return 'quick';
}

async function addCommand(project: string, requirement: string, opts: AddOptions): Promise<void> {
  const config = getConfig();
  const scope = opts.as ?? detectScope(requirement);

  let description = requirement;
  let requirementPath: string | null = null;

  if (isFilePath(requirement)) {
    requirementPath = path.resolve(requirement);
    const content = readFileSync(requirementPath, 'utf8');
    // Extract title from first markdown heading
    const title = content.match(/^#\s+(.+)/m)?.[1];
    description = title ?? path.basename(requirement, '.md');
  } else if (isDirPath(requirement)) {
    requirementPath = path.resolve(requirement);
    description = path.basename(requirementPath);
  }

  // For quick scope with file, pass FULL content as description (v1 bug fix)
  if (scope === 'quick' && isFilePath(requirement)) {
    description = readFileSync(path.resolve(requirement), 'utf8');
  }

  const job = addJob(project, scope, description, requirementPath);

  if (isJsonMode()) {
    outputJson({ job });
    return;
  }

  outputHuman(`  ${green('✓')} Queued: ${project} · ${scope} · "${description.slice(0, 60)}"  ${dim(`(id: ${job.id})`)}`);
  
  // TODO: check daemon status when service command implemented
  outputHuman(`  ${dim('Run:')} pilot service start ${dim('to process queue')}`);
}

export { addCommand };
```

```typescript
import { getConfig } from '../core/config.js';

import type { Job } from '../core/types.js';

interface StatusOptions {
  json?: boolean;
}

async function statusCommand(opts: StatusOptions): Promise<void> {
  const queue = getQueue();
  const recent = getRecent(10);

  const active = queue.filter(j => j.status === 'running');
  const pending = queue.filter(j => j.status === 'pending');

  if (isJsonMode()) {
    outputJson({
      version: '2.0.0', // from package.json eventually
      daemon: { active: false }, // TODO: detect daemon
      active,
      queue: pending,
      recent,
    });
    return;
  }

  // Header
  outputHuman('');
  outputHuman(`  ${bold('pilot')} v2.0.0`);
  outputHuman('');

  // Active section
  if (active.length > 0) {
    outputHuman(`  ${bold('Active')} (${active.length})`);
    for (const job of active) {
      const elapsed = job.startedAt ? formatRelativeTime(job.startedAt) : '';
      outputHuman(`  ${blue('●')} ${job.project}  ${dim(job.scope)}  "${job.description.slice(0, 50)}"  ${dim(elapsed)}`);
      
      // Try to get session details from opencode DB
      // Use session title pattern: {project}-{command}-{id}
      // This enriches the display but isn't critical if unavailable
    }
    outputHuman('');
  }

  // Queue section
  if (pending.length > 0) {
    outputHuman(`  ${bold('Queue')} (${pending.length})`);
    for (const job of pending) {
      outputHuman(`  ${dim('○')} ${job.project}  ${dim(job.scope)}  "${job.description.slice(0, 50)}"  ${dim('pending')}`);
    }
    outputHuman('');
  }

  // Recent section
  if (recent.length > 0) {
    outputHuman(`  ${bold('Recent')}`);
    for (const job of recent) {
      const icon = job.status === 'completed' ? green('✓') : job.status === 'failed' ? red('✗') : dim('◌');
      const elapsed = job.completedAt ? formatRelativeTime(job.completedAt) : '';
      outputHuman(`  ${icon} ${job.project}  ${dim(job.scope)}  "${job.description.slice(0, 50)}"  ${dim(elapsed)}`);
    }
    outputHuman('');
  }

  if (active.length === 0 && pending.length === 0 && recent.length === 0) {
    outputHuman(`  ${dim('No jobs. Run: pilot add <project> <requirement>')}`);
    outputHuman('');
  }
}

export { statusCommand };
```

```typescript
import { getQueue, getRecent } from '../core/db.js';
import { outputJson, outputHuman, isJsonMode } from '../util/output.js';
import { bold, dim, green, red, blue, cyan } from '../util/colors.js';
import { formatRelativeTime, formatDuration } from '../util/format.js';

interface QueueOptions {
  json?: boolean;
  history?: boolean;
}

async function queueCommand(opts: QueueOptions): Promise<void> {
  if (opts.history) {
    const recent = getRecent(50);
    
    if (isJsonMode()) {
      outputJson({ history: recent });
      return;
    }
    
    outputHuman('');
    outputHuman(`  ${bold('History')} (last ${recent.length})`);
    outputHuman('');
    
    for (const job of recent) {
      const icon = job.status === 'completed' ? green('✓') : job.status === 'failed' ? red('✗') : dim('◌');
      const elapsed = job.completedAt ? formatRelativeTime(job.completedAt) : '';
      outputHuman(`  ${icon}  ${dim(job.id)}  ${job.project}  ${dim(job.scope)}  ${job.description.slice(0, 40)}  ${dim(elapsed)}`);
    }
    
    outputHuman('');
    return;
  }

  const queue = getQueue();
  
  if (isJsonMode()) {
    outputJson({ queue });
    return;
  }

  outputHuman('');
  
  if (queue.length === 0) {
    outputHuman(`  ${dim('Queue empty')}`);
    outputHuman('');
    return;
  }

  outputHuman(`  ${bold('#')}   ${bold('project')}    ${bold('scope')}     ${bold('description')}                         ${bold('status')}`);
  
  for (const [i, job] of queue.entries()) {
    const icon = job.status === 'running' ? blue('●') : dim('○');
    const status = job.status === 'running' 
      ? `${blue('running')} ${dim(job.startedAt ? formatRelativeTime(job.startedAt) : '')}`
      : dim('pending');
    outputHuman(`  ${icon} ${dim(job.id)}  ${job.project.padEnd(10)}  ${dim(job.scope.padEnd(9))}  ${job.description.slice(0, 35).padEnd(35)}  ${status}`);
  }
  
  outputHuman('');
}

export { queueCommand };
```

```typescript
import { getJob, getQueue } from '../core/db.js';
import { findSessionByTitle, getSessionMessages, exportSessionFromDb } from '../core/opencode-db.js';
import { outputJson, outputHuman, isJsonMode } from '../util/output.js';
import { bold, dim, cyan, green, yellow } from '../util/colors.js';
import type { SessionMessage } from '../core/types.js';

interface LogOptions {
  json?: boolean;
  follow?: boolean;
  last?: number;
}

async function logCommand(idOrUndefined: string | undefined, opts: LogOptions): Promise<void> {
  let jobId = idOrUndefined;

  // Smart default: if no ID, show latest running job
  if (!jobId) {
    const queue = getQueue();
    const running = queue.find(j => j.status === 'running');
    if (running) {
      jobId = running.id;
    } else {
      process.stderr.write('No running jobs. Specify a job ID: pilot log <id>\n');
      process.exit(1);
    }
  }

  const job = getJob(jobId);
  if (!job) {
    process.stderr.write(`Job not found: ${jobId}\n`);
    process.exit(1);
  }

  // Find the session(s) for this job
  const sessionTitles: string[] = job.sessionTitles ? JSON.parse(job.sessionTitles) : [];
  
  // If no stored titles, try to find by title pattern
  if (sessionTitles.length === 0) {
    // Convention: session titles are {project}-{command}-{jobId}
    // We search for any session containing the job ID
  }

  // Get all messages from all sessions for this job
  let allMessages: SessionMessage[] = [];
  
  for (const title of sessionTitles) {
    const sessionId = findSessionByTitle(title);
    if (sessionId) {
      const messages = getSessionMessages(sessionId);
      allMessages = allMessages.concat(messages);
    }
  }

  // Also try the latest delegation/step sessions by pattern
  // This handles cases where sessionTitles hasn't been populated yet

  if (isJsonMode()) {
    outputJson({
      job: { id: job.id, project: job.project, scope: job.scope, description: job.description },
      messages: allMessages,
    });
    return;
  }

  // Header
  outputHuman('');
  outputHuman(`  ${bold(job.project)} · ${job.scope} · "${job.description.slice(0, 50)}" · ${dim(job.id)}`);
  outputHuman('');

  if (allMessages.length === 0) {
    outputHuman(`  ${dim('No messages yet')}`);
    outputHuman('');
    return;
  }

  // Apply --last filter
  const messages = opts.last ? allMessages.slice(-opts.last) : allMessages;

  for (const msg of messages) {
    const time = new Date(msg.createdAt).toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' });
    const role = msg.role === 'assistant' ? cyan('[assistant]') : msg.role === 'user' ? green('[user]') : yellow(`[${msg.role}]`);
    const content = msg.content.slice(0, 200);
    outputHuman(`  ${dim(time)}  ${role} ${content}`);
  }

  outputHuman('');

  // --follow mode: poll for new messages
  if (opts.follow) {
    outputHuman(dim('  Following... (Ctrl-C to stop)'));
    let lastSeen = allMessages.length > 0 ? allMessages[allMessages.length - 1].createdAt : 0;
    
    const poll = async () => {
      while (true) {
        await new Promise(r => setTimeout(r, 500));
        
        for (const title of sessionTitles) {
          const sessionId = findSessionByTitle(title);
          if (!sessionId) continue;
          
          const newMsgs = getSessionMessages(sessionId, lastSeen);
          for (const msg of newMsgs) {
            const time = new Date(msg.createdAt).toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' });
            const role = msg.role === 'assistant' ? cyan('[assistant]') : msg.role === 'user' ? green('[user]') : yellow(`[${msg.role}]`);
            outputHuman(`  ${dim(time)}  ${role} ${msg.content.slice(0, 200)}`);
            lastSeen = msg.createdAt;
          }
        }
      }
    };
    
    await poll();
  }
}

export { logCommand };
```
