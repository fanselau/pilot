import { describe, it, expect, vi, beforeEach } from 'vitest';

vi.mock('../../src/core/db.js', () => ({
  getJob: vi.fn(),
  cancel: vi.fn(),
  retry: vi.fn(),
  bump: vi.fn(),
}));

// Mock process.exit to prevent test termination
const mockExit = vi.spyOn(process, 'exit').mockImplementation(() => { throw new Error('exit'); });
const mockStderr = vi.spyOn(process.stderr, 'write').mockImplementation(() => true);

import { cancelCommand } from '../../src/commands/cancel.js';
import { retryCommand } from '../../src/commands/retry.js';
import { bumpCommand } from '../../src/commands/bump.js';
import { getJob, cancel, retry, bump } from '../../src/core/db.js';

describe('cancel', () => {
  beforeEach(() => { vi.clearAllMocks(); });

  it('cancels pending job', async () => {
    vi.mocked(getJob).mockReturnValue({ id: 'abc1', status: 'pending' } as any);
    await cancelCommand('abc1');
    expect(cancel).toHaveBeenCalledWith('abc1');
  });

  it('rejects running job', async () => {
    vi.mocked(getJob).mockReturnValue({ id: 'abc1', status: 'running' } as any);
    await expect(cancelCommand('abc1')).rejects.toThrow('exit');
    expect(cancel).not.toHaveBeenCalled();
  });
});
