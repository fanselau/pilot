The flow:

```typescript
// After owner registration block, before config init trigger:

// ── Optional: offer recommended skills ──────────────────────────────
if (result.errors.length === 0 && !isJsonMode()) {
  try {
    const { recommendDefaultSkills, bootstrapDefaultSkills } = await import('../core/default-skills.js');
    const absDir = path.resolve(dir);
    const recommendation = recommendDefaultSkills(absDir);
    
    if (recommendation.skills.length > 0) {
      outputHuman('');
      
      // Show detected stack
      if (recommendation.detectedStack.items.length > 0) {
        outputHuman(`  ${dim('Detected stack:')} ${recommendation.detectedStack.items.join(', ')}`);
      }
      
      outputHuman(`  ${dim(`${recommendation.skills.length} recommended skills available`)}`);
      
      // Only prompt if TTY
      if (process.stdin.isTTY) {
        const readline = await import('node:readline');
        const rl = readline.createInterface({ input: process.stdin, output: process.stdout });
        
        const answer = await new Promise<string>(resolve => {
          rl.question(`  Install recommended skills? (Y/n) `, resolve);
        });
        rl.close();
        
        if (answer.toLowerCase() !== 'n') {
          const result = await bootstrapDefaultSkills({ projectDir: absDir, yes: true });
          
          outputHuman('');
          for (const err of result.errors) {
            outputHuman(`  ${yellow('⚠')} ${err.skill}: ${dim(err.error)}`);
          }
          if (result.installed > 0) {
            outputHuman(`  ${green('✓')} Installed ${result.installed} skills`);
          }
          if (result.failed > 0) {
            outputHuman(`  ${dim(`${result.failed} failed (see warnings above)`)}`);
          }
        }
      } else {
        outputHuman(`  ${dim('Run `pilot skills bootstrap --yes` to install')}`);
      }
    }
  } catch (err) {
    // Non-fatal: skill bootstrap failure must NEVER make setup fail
    // Just silently skip — setup already succeeded
    process.stderr.write(`Warning: skill recommendation failed: ${err}\n`);
  }
}
```

**Critical guardrails:**
- The entire skill block is wrapped in try/catch — if anything goes wrong (import fails, detection fails, install fails), setup still reports success
- Non-TTY: show hint to run manually, don't try to prompt
- JSON mode: skip entirely (already gated by `!isJsonMode()`)
- Do NOT shadow the outer `result` variable — use a different name like `bootstrapResult` for the bootstrap return value
- The prompt defaults to Yes (empty answer → install). Only explicit 'n' or 'N' skips.

**Import additions:**
- `yellow` needs to be added to the color imports at the top (it's already imported, verify)
- `green`, `dim` are already imported
