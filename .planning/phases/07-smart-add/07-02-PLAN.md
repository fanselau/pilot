**Completely rewrite `src/commands/add.ts`** — replace the old mode-based command.

New signature: `addCommand(project: string, input: string, opts: Record<string, unknown>)`

Where `input` is either:
- A file path (e.g., `requirements/dark-mode.md`) — detected by checking if file exists
- A directory path (e.g., `requirements/v2/`) — detected by checking if directory exists
- A string description (e.g., `"fix the favicon"`) — fallback if neither file nor dir

**Implementation flow:**

```typescript
import { access, stat, readFile, readdir, mkdir, writeFile, copyFile } from 'node:fs/promises';
import path from 'node:path';
import { getConfig } from '../core/config.js';
import { detectScope, detectProjectState, generateRequirementsContent, resolveInternalMode, parseRequirementsFile } from '../core/smart-add.js';
import { addItem, getItems } from '../core/queue-store.js';
import { setupProject } from '../core/setup.js';
import { isJsonMode, outputJson, outputHuman } from '../util/output.js';
import { dim, yellow } from '../util/colors.js';
import type { SmartAddScope } from '../core/types.js';

export async function addCommand(
  project: string,
  input: string,
  opts: Record<string, unknown>,
): Promise<void> {
  const config = getConfig();
  const projectDir = path.join(config.projectDir, project);

  // 1. Determine input type: file, directory, or description
  let content: string | null = null;
  let isDirectory = false;
  let requirementsPath: string | null = null;
  let description = input;

  // Check if input is a file path
  try {
    const inputPath = path.resolve(input);
    const inputStat = await stat(inputPath);
    if (inputStat.isDirectory()) {
      isDirectory = true;
      requirementsPath = inputPath;
      // Count files in directory for description
      const files = await readdir(inputPath);
      const mdFiles = files.filter(f => f.endsWith('.md'));
      description = `${mdFiles.length} requirements files from ${path.basename(inputPath)}/`;
    } else if (inputStat.isFile()) {
      content = await readFile(inputPath, 'utf8');
      requirementsPath = inputPath;
      // Use first heading or filename as description
      const headingMatch = content.match(/^#\s+(.+)/m);
      description = headingMatch ? headingMatch[1] : path.basename(inputPath, '.md');
    }
  } catch {
    // Not a file/dir path — treat as string description
    content = null;
    requirementsPath = null;
    description = input;
  }

  // 2. Scope override via --as flag
  let scopeOverride: SmartAddScope | null = null;
  if (opts['as'] && typeof opts['as'] === 'string') {
    const valid: SmartAddScope[] = ['quick', 'phase', 'milestone'];
    if (!valid.includes(opts['as'] as SmartAddScope)) {
      process.stderr.write(`Error: Invalid scope '${opts['as']}'. Valid: quick, phase, milestone\n`);
      process.exit(2);
    }
    scopeOverride = opts['as'] as SmartAddScope;
  }

  // 3. Detect scope
  const scopeResult = scopeOverride
    ? { scope: scopeOverride, itemCount: 0, hasPhaseHeaders: false, isDirectory, rationale: `Override: --as ${scopeOverride}` }
    : detectScope({ content, description, isDirectory });

  // 4. Detect project state
  const projectState = await detectProjectState(project, config);

  // 5. Handle missing project directory
  if (!projectState.exists) {
    process.stderr.write(`Error: Project '${project}' not found in ${config.projectDir}/\n`);
    process.exit(1);
  }

  // 6. Auto-setup if needed
  if (projectState.needsSetup) {
    if (!isJsonMode()) {
      outputHuman(`${dim('→')} Running setup for ${project}...`);
    }
    await setupProject(projectDir);
  }

  // 7. Warn if already queued or running
  if (projectState.isQueued && !isJsonMode()) {
    outputHuman(yellow(`Warning: ${project} already queued (${projectState.queuedMode}). Adding anyway.`));
  }
  if (projectState.isRunning && !isJsonMode()) {
    outputHuman(yellow(`Warning: ${project} is currently running${projectState.runningPhase ? ` (${projectState.runningPhase})` : ''}. Queuing after completion.`));
  }

  // 8. Handle requirements file — copy to project if not already there
  if (requirementsPath && !isDirectory) {
    const projectReqDir = path.join(projectDir, 'requirements');
    const targetPath = path.join(projectReqDir, path.basename(requirementsPath));
    // Only copy if source is outside the project dir
    if (!requirementsPath.startsWith(projectDir)) {
      await mkdir(projectReqDir, { recursive: true });
      await copyFile(requirementsPath, targetPath);
      if (!isJsonMode()) {
        outputHuman(`${dim('→')} Copied ${path.basename(requirementsPath)} to ${project}/requirements/`);
      }
    }
  }

  // 9. Handle string description — generate requirements file
  if (!requirementsPath && scopeResult.scope !== 'quick') {
    const reqContent = generateRequirementsContent(description);
    const projectReqDir = path.join(projectDir, 'requirements');
    const safeName = description.slice(0, 40).replace(/[^a-zA-Z0-9]+/g, '-').replace(/^-|-$/g, '').toLowerCase();
    const targetPath = path.join(projectReqDir, `${safeName}.md`);
    await mkdir(projectReqDir, { recursive: true });
    await writeFile(targetPath, reqContent);
    requirementsPath = targetPath;
    if (!isJsonMode()) {
      outputHuman(`${dim('→')} Generated requirements: ${project}/requirements/${safeName}.md`);
    }
  }

  // 10. Resolve internal mode
  const internalMode = resolveInternalMode(scopeResult.scope, projectState);

  // 11. Log the decision
  if (!isJsonMode()) {
    outputHuman(`${dim('→')} Detected: ${scopeResult.scope} (${scopeResult.rationale})`);
  }

  // 12. Dry-run — show what would happen without queuing
  if (opts['dryRun'] === true) {
    if (isJsonMode()) {
      outputJson({
        action: 'dry-run',
        project,
        scope: scopeResult.scope,
        internalMode,
        description,
        requirementsPath,
        projectState: {
          needsSetup: projectState.needsSetup,
          needsInit: projectState.needsInit,
          isQueued: projectState.isQueued,
          isRunning: projectState.isRunning,
        },
        scopeDetection: scopeResult,
      });
    } else {
      outputHuman(`Dry run — would queue: ${project} | ${internalMode} | ${description}`);
    }
    return;
  }

  // 13. Add to queue
  const id = await addItem({
    project,
    mode: internalMode,
    description,
  });

  // 14. Output
  if (isJsonMode()) {
    outputJson({
      action: 'added',
      id,
      project,
      scope: scopeResult.scope,
      internalMode,
      description,
      requirementsPath,
    });
  } else {
    outputHuman(`✓ Queued (${id}): ${project} | ${scopeResult.scope} → ${internalMode}`);
    if (requirementsPath) {
      outputHuman(`  Requirements: ${requirementsPath}`);
    }
    outputHuman(`  Run \`pilot run\` to start building`);
  }
}
```

**Key design decisions:**
- `input` is always a single string (file path, dir path, or description) — commander captures it as one arg
- File/dir detection via `stat()` with try/catch fallback to string description
- Scope override via `--as` flag (nice-to-have from spec, trivial to add)
- Auto-setup runs silently if .opencode/ is missing
- Warnings for queued/running are non-blocking (auto-yes in non-interactive, per spec)
- Requirements file is copied to project dir if external (so GSD agents can find it)
- For non-quick scopes with only a string description, generate a minimal requirements file

**Rewrite `src/commands/build.ts`** — becomes `add + run` convenience:

```typescript
/**
 * pilot build <project> <requirement-or-description> — Smart add + run.
 *
 * Convenience: calls addCommand then starts runner if not active.
 */

import { execa } from 'execa';
import { getConfig } from '../core/config.js';
import { readPidFile, isProcessAlive } from '../core/process.js';
import { isJsonMode, outputJson, outputHuman } from '../util/output.js';
import { addCommand } from './add.js';

export async function buildCommand(
  project: string,
  input: string | undefined,
  opts: Record<string, unknown>,
): Promise<void> {
  // Default input to empty description if not provided
  const addInput = input ?? '';

  if (!addInput) {
    process.stderr.write('Error: Provide a requirements file or description\n');
    process.exit(2);
  }

  // Run smart add (queues the work)
  await addCommand(project, addInput, opts);

  // Unless --no-run, check if runner is active and start if not
  const shouldRun = opts['run'] !== false;
  let runnerStarted = false;

  if (shouldRun) {
    const runnerPid = await readPidFile('queue');
    const runnerAlive = runnerPid !== null && isProcessAlive(runnerPid);

    if (runnerAlive) {
      if (!isJsonMode()) {
        outputHuman(`Runner already active (PID ${runnerPid})`);
      }
    } else {
      try {
        const pilotBin = process.argv[1]!;
        const child = execa('node', [pilotBin, 'run'], {
          detached: true,
          stdin: 'ignore',
          stdout: 'ignore',
          stderr: 'ignore',
        });
        child.unref();
        runnerStarted = true;

        if (!isJsonMode()) {
          outputHuman('✓ Runner started');
        }
      } catch (err) {
        if (!isJsonMode()) {
          outputHuman(`Warning: Failed to start runner: ${err instanceof Error ? err.message : String(err)}`);
        }
      }
    }
  }

  if (isJsonMode() && runnerStarted) {
    // addCommand already output JSON — but we can note runner separately
    // Actually, addCommand already handles JSON output. We just need to handle the runner part.
    // Since addCommand already wrote JSON to stdout, don't double-output.
    // Only output runner status if it's meaningful
  }
}
```
