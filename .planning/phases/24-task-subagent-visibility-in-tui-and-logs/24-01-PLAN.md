```typescript
function getChildSessions(parentSessionId: string): Array<{ id: string; title: string; timeCreated: number; timeUpdated: number }> {
  const db = openDb();
  if (db === null) return [];

  try {
    const rows = db.prepare(
      'SELECT id, title, time_created, time_updated FROM session WHERE parent_id = ? ORDER BY time_created ASC',
    ).all(parentSessionId) as Array<{ id: string; title: string; time_created: number; time_updated: number }>;

    return rows.map((row) => ({
      id: row.id,
      title: row.title,
      timeCreated: row.time_created,
      timeUpdated: row.time_updated,
    }));
  } catch {
    return [];
  }
}
```

```json
{
```

Extract `description` and `subagent_type` and return:
```
▶ task: gsd-planner — "Quick plan for inline editing"
```

If `subagent_type` is missing, use `"subagent"`. If `description` is missing, truncate the full JSON to 100 chars.

Add this case in `extractToolInput` BEFORE the `if (tool === 'bash')` block:

```typescript
if (tool === 'task') {
  if (typeof stateInput === 'object' && stateInput !== null) {
    const inp = stateInput as Record<string, unknown>;
    const desc = typeof inp.description === 'string' ? inp.description : null;
    const agentType = typeof inp.subagent_type === 'string' ? inp.subagent_type : 'subagent';
    if (desc) {
      return `▶ task: ${agentType} — "${truncateStr(desc, 80)}"`;
    }
  }
  return truncateStr(typeof stateInput === 'string' ? stateInput : JSON.stringify(stateInput), 100);
}
```

This means any consumer that calls `part.toolInput` for a task part will see the formatted string automatically.
