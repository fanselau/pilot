  4. **Add DB helper functions** in db.ts:

     ```typescript
     export function incrementHungCount(jobId: string, hungReason: string): void {
       getDb().prepare(`
         UPDATE jobs SET hung_count = hung_count + 1, last_hung_reason = ?
         WHERE id = ?
       `).run(hungReason, jobId);
     }

     export function incrementRetryCount(jobId: string): void {
       getDb().prepare(`
         UPDATE jobs SET retry_count = retry_count + 1
         WHERE id = ?
       `).run(jobId);
     }

     export function canRetry(jobId: string): boolean {
       const row = getDb().prepare(`
         SELECT retry_count, retry_budget FROM jobs WHERE id = ?
       `).get(jobId) as { retry_count: number; retry_budget: number } | undefined;
       if (!row) return false;
       return row.retry_count < row.retry_budget;
     }

     export function isSameHungReason(jobId: string, currentReason: string): boolean {
       const row = getDb().prepare(`
         SELECT last_hung_reason FROM jobs WHERE id = ?
       `).get(jobId) as { last_hung_reason: string | null } | undefined;
       return row?.last_hung_reason === currentReason;
     }

     export function resetRetryState(jobId: string): void {
       getDb().prepare(`
         UPDATE jobs SET retry_count = 0, hung_count = 0, last_hung_reason = NULL
         WHERE id = ?
       `).run(jobId);
     }
     ```

     } catch (err) {
       if (err instanceof HungSessionError) {
         // Track hung count
         incrementHungCount(job.id, err.hungReason);

         // Same-error escalation: two consecutive hangs with same reason → immediate fail
         if (isSameHungReason(job.id, err.hungReason) && job.hungCount >= 1) {
           this.log(`Same hung reason repeated for ${job.id}: ${err.hungReason} — escalating to failure`);
           markFailed(job.id, `Escalated: consecutive ${err.hungReason} hangs (tool: ${err.lastToolCall ?? 'unknown'})`);
           // Notify on escalation (budget exhaustion equivalent)
           const failedJob = getJob(job.id);
           if (failedJob) await notifyJobCompletion(failedJob);
           return;
         }

         // Check retry budget
         if (canRetry(job.id)) {
           incrementRetryCount(job.id);
           this.log(`Hung retry ${job.retryCount + 1}/${job.retryBudget} for ${job.id}: ${err.hungReason}`);

           // Reset to pending for re-launch
           // For interactive-prompt on phase jobs: use --gaps if there was progress
           const hint = err.hungReason === 'interactive-prompt' ? 'gaps-if-progress' : null;
           resetToPending(job.id, `Hung retry: ${err.hungReason} (tool: ${err.lastToolCall ?? 'unknown'})`, hint);
           // Do NOT notify — retry is silent
           return;
         }

         // Budget exhausted
         this.log(`Retry budget exhausted for ${job.id} (${job.retryBudget} retries used)`);
         markFailed(job.id, `Retry budget exhausted after ${err.hungReason} hang (tool: ${err.lastToolCall ?? 'unknown'})`);
         const failedJob = getJob(job.id);
         if (failedJob) await notifyJobCompletion(failedJob);
         return;
       }

       // Existing catch-all for other errors...
       const error = errMsg(err);
       // ... existing markFailed + notify logic
     }
