**Shared helper (can be in a util file or inlined in each command — prefer inlined to avoid new file):**

```typescript
import { execa } from 'execa';
import path from 'node:path';
import { getConfig } from '../core/config.js';

function resolveProjectDir(project: string): string {
  const config = getConfig();
  return path.join(config.projectDir, project);
}

function truncateTitle(title: string, max = 80): string {
  return title.length > max ? title.slice(0, max) : title;
}

function sanitizeArgs(args: string): string {
  return args.replace(/[^a-zA-Z0-9-]/g, '-').replace(/-+/g, '-').replace(/^-|-$/g, '');
}
```

**Pattern for each lifecycle command:**

```typescript
import { execa } from 'execa';
import path from 'node:path';
import { getConfig } from '../core/config.js';

export async function planCommand(
  project: string,
  phase: string,
  opts: Record<string, unknown>,
): Promise<void> {
  const dir = path.join(getConfig().projectDir, project);
  const gsdCommand = 'gsd-plan-phase';
  
  // Build args string
  let args = `${phase} --auto`;
  if (opts.research) args += ' --research';
  if (opts.skipResearch) args += ' --skip-research';
  if (opts.gaps) args += ' --gaps';
  
  const title = truncateTitle(`${project}-${gsdCommand}-${sanitize(args)}`);
  
  const result = await execa('opencode', [
    'run', '--format', 'default', '--title', title,
    '--command', gsdCommand,
    ...(args.includes('--') ? ['--', args] : [args])
  ], { cwd: dir, stdio: 'inherit' });
  
  process.exit(result.exitCode ?? 0);
}
```

**src/commands/init.ts** — pilot init is special: it creates the project before spawning.

Per spec §5:
1. Resolve project dir from PILOT_PROJECT_DIR + project name
2. Create project dir if it doesn't exist (mkdir -p)
3. Run `pilot setup <project-dir>` if `.opencode/` doesn't exist — import and call setupProject from core/setup.ts directly (no need to spawn a separate process)
4. Spawn `opencode run --command gsd-new-project [--auto] [description]` in the project dir with stdio: 'inherit'
5. If opts.auto is set, add `--auto` flag

```typescript
import { execa } from 'execa';
import { mkdir } from 'node:fs/promises';
import { access } from 'node:fs/promises';
import path from 'node:path';
import { getConfig } from '../core/config.js';
import { setupProject } from '../core/setup.js';

export async function initCommand(
  project: string,
  desc: string | undefined,
  opts: Record<string, unknown>,
): Promise<void> {
  const config = getConfig();
  const projectDir = path.join(config.projectDir, project);
  
  // Create dir if needed
  await mkdir(projectDir, { recursive: true });
  
  // Setup if .opencode/ doesn't exist
  try {
    await access(path.join(projectDir, '.opencode'));
  } catch {
    await setupProject(projectDir);
  }
  
  // Build args
  const gsdArgs: string[] = [];
  if (opts.auto) gsdArgs.push('--auto');
  if (desc) gsdArgs.push(desc);
  
  const title = truncateTitle(`${project}-new-project`);
  
  const result = await execa('opencode', [
    'run', '--format', 'default', '--title', title,
    '--command', 'gsd-new-project',
    ...(gsdArgs.length > 0 ? gsdArgs : [])
  ], { cwd: projectDir, stdio: 'inherit' });
  
  process.exit(result.exitCode ?? 0);
}
```
