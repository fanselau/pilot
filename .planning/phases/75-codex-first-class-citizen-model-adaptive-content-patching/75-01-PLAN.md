**3. adaptContent(content: string, family: 'codex'): string:**

Apply transformations as a pipeline of regex/string replacements. Each rule is a separate function for testability:

**Rule 1 — Identity neutralization:**
- Replace "Claude" when used as an identity/actor — at start of sentence, after articles ("the Claude", "as Claude", "You are Claude"), in phrases like "Claude Code"
- Pattern: `/(?<![`\w])Claude(?![`\w.-])/g` — matches "Claude" NOT preceded by backtick or word char, NOT followed by backtick, word char, dot, or hyphen
- But be more precise: skip lines that contain backtick-wrapped model identifiers. Use a line-by-line approach:
  - For each line, if line contains a backtick-wrapped string with "claude" (e.g. `` `claude-opus-4-6` ``), skip identity replacement on that line
  - Otherwise replace "Claude Code" → "the coding agent", "Claude" (standalone as subject) → "the agent" or "You" depending on sentence position
- Simpler approach: use two-pass:
  - First: replace "Claude Code" (as a tool/runtime name) → "the coding agent"
  - Then: replace standalone "Claude" as actor → "You" when at sentence start, "the agent" elsewhere
  - Skip any occurrence inside backticks using negative lookbehind/lookahead for backtick

Concrete regexes:
```typescript
// Pass 1: "Claude Code" as runtime name (not in backticks)
content = content.replace(/(?<!`)Claude Code(?!`)/g, 'the coding agent');

// Pass 2: "You are Claude" patterns → "You are the agent"
content = content.replace(/You are Claude/g, 'You are the agent');

// Pass 3: "Claude" at start of sentence → "You" 
// Matches: period/newline + optional whitespace + "Claude" + space
content = content.replace(/((?:^|[.!?]\s+))Claude(?=\s)/gm, '$1You');

// Pass 4: Remaining standalone "Claude" (not in backticks, not part of a path/model-id) → "the agent"
// Negative lookbehind for ` / . - _ (part of identifiers)
// Negative lookahead for ` . - _ / (part of identifiers)
content = content.replace(/(?<![`\/._-])Claude(?![`\/._-])/g, 'the agent');
```

**Rule 2 — Tool reference generalization:**
```typescript
// "Bash tool" / "the Bash tool" → "the terminal"
content = content.replace(/(?:the\s+)?Bash tool/gi, 'the terminal');
// "Write tool" / "the Write tool" → "the file writing tool"
content = content.replace(/(?:the\s+)?Write tool/gi, 'the file writing tool');
```

**5. getFilesToPatch(projectDir: string, providerMode: string): string[]:**

```typescript
import { readdirSync, existsSync } from 'node:fs';
import path from 'node:path';

export function getFilesToPatch(projectDir: string, providerMode: string): string[] {
  if (providerMode === 'claude-only') return [];

  const files: string[] = [];

  // Agent files
  const agentsDir = path.join(projectDir, '.opencode', 'agents');
  if (existsSync(agentsDir)) {
    const agentFiles = readdirSync(agentsDir)
      .filter(f => f.endsWith('.md'))
      .map(f => path.join(agentsDir, f));

    if (providerMode === 'hybrid') {
      // Only check-role agents
      files.push(...agentFiles.filter(f => {
        const name = path.basename(f, '.md');
        return isCheckRoleAgent(name);
      }));
    } else {
      // openai-only: all agents
      files.push(...agentFiles);
    }
  }

  // For openai-only mode, also patch commands, workflows, references
  if (providerMode !== 'hybrid') {
    const commandDir = path.join(projectDir, '.opencode', 'command');
    if (existsSync(commandDir)) {
      files.push(...readdirSync(commandDir).filter(f => f.endsWith('.md')).map(f => path.join(commandDir, f)));
    }

    const workflowDir = path.join(projectDir, '.opencode', 'get-shit-done', 'workflows');
    if (existsSync(workflowDir)) {
      files.push(...readdirSync(workflowDir).filter(f => f.endsWith('.md')).map(f => path.join(workflowDir, f)));
    }

    const refDir = path.join(projectDir, '.opencode', 'get-shit-done', 'references');
    if (existsSync(refDir)) {
      files.push(...readdirSync(refDir).filter(f => f.endsWith('.md')).map(f => path.join(refDir, f)));
    }
  }

  return files;
}
```

**6. patchContentForModel(projectDir, providerMode, models):**

```typescript
import { readFileSync, writeFileSync } from 'node:fs';

export function patchContentForModel(
  projectDir: string,
  providerMode: string,
  models: Record<string, { model: string; variant?: string }>,
): PatchState {
  const state: PatchState = {
    originals: new Map(),
    projectDir,
  };

  // Determine the target model family from any model entry
  const anyModel = Object.values(models)[0];
  if (!anyModel) return state;
  const family = getModelFamily(anyModel.model);
  if (family !== 'codex') return state; // Only adapt for Codex

  const files = getFilesToPatch(projectDir, providerMode);
  for (const filePath of files) {
    try {
      const original = readFileSync(filePath, 'utf8');
      state.originals.set(filePath, original);
      const adapted = adaptContent(original, 'codex');
      if (adapted !== original) {
        writeFileSync(filePath, adapted, 'utf8');
      }
    } catch {
      // Skip unreadable files — don't break the spawn
    }
  }

  return state;
}
```

**7. restoreContent(patchState):**

```typescript
export function restoreContent(patchState: PatchState): void {
  for (const [filePath, original] of patchState.originals) {
    try {
      writeFileSync(filePath, original, 'utf8');
    } catch {
      process.stderr.write(`[prompt-adapter] Warning: failed to restore ${filePath}\n`);
    }
  }
}
```
