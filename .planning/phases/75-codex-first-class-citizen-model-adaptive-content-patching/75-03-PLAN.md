**1. adaptDryRunCommand(projectPath: string, options: { provider?: string }):**

This shows what `patchContentForModel` WOULD do without modifying any files.

```typescript
export async function adaptDryRunCommand(
  projectPath: string,
  options: { provider?: string },
): Promise<void> {
  const projectDir = resolveProjectDir(projectPath);
  const providerMode = options.provider ?? 'openai-only';

  const files = getFilesToPatch(projectDir, providerMode);
  if (files.length === 0) {
    process.stdout.write(`No files to patch for provider mode: ${providerMode}\n`);
    return;
  }

  process.stdout.write(`Dry run: ${files.length} file(s) would be patched for ${providerMode} mode\n\n`);

  let totalReplacements = 0;
  for (const filePath of files) {
    const relativePath = path.relative(projectDir, filePath);
    try {
      const original = readFileSync(filePath, 'utf8');
      const adapted = adaptContent(original, 'codex');
      if (adapted === original) {
        process.stdout.write(`  ${relativePath}: no changes\n`);
        continue;
      }

      // Count differences (simple line-level diff)
      const origLines = original.split('\n');
      const adaptLines = adapted.split('\n');
      let changedLines = 0;
      const maxLen = Math.max(origLines.length, adaptLines.length);
      const diffs: Array<{ line: number; before: string; after: string }> = [];
      for (let i = 0; i < maxLen; i++) {
        if (origLines[i] !== adaptLines[i]) {
          changedLines++;
          if (diffs.length < 10) { // Show first 10 diffs per file
            diffs.push({ line: i + 1, before: origLines[i] ?? '', after: adaptLines[i] ?? '' });
          }
        }
      }

      process.stdout.write(`  ${relativePath}: ${changedLines} line(s) changed\n`);
      totalReplacements += changedLines;
      for (const d of diffs) {
        process.stdout.write(`    L${d.line}: - ${d.before.trim()}\n`);
        process.stdout.write(`    L${d.line}: + ${d.after.trim()}\n`);
      }
      if (changedLines > diffs.length) {
        process.stdout.write(`    ... and ${changedLines - diffs.length} more\n`);
      }
      process.stdout.write('\n');
    } catch (err) {
      process.stdout.write(`  ${relativePath}: error reading file\n`);
    }
  }

  process.stdout.write(`\nTotal: ${totalReplacements} line(s) across ${files.length} file(s)\n`);
}
```

**2. adaptScanCommand(projectPath: string):**

Scans all GSD files for "Claude" occurrences and classifies each as identity/model-id/filename/doc.

```typescript
export async function adaptScanCommand(projectPath: string): Promise<void> {
  const projectDir = resolveProjectDir(projectPath);

  // Scan ALL .md files under .opencode/ (not mode-filtered)
  const allFiles = getFilesToPatch(projectDir, 'openai-only'); // Gets maximum file set

  if (allFiles.length === 0) {
    process.stdout.write('No GSD files found in project\n');
    return;
  }

  process.stdout.write(`Scanning ${allFiles.length} file(s) for Claude references...\n\n`);

  let totalMatches = 0;
  for (const filePath of allFiles) {
    const relativePath = path.relative(projectDir, filePath);
    try {
      const content = readFileSync(filePath, 'utf8');
      const lines = content.split('\n');

      const matches: Array<{ line: number; text: string; classification: string }> = [];
      for (let i = 0; i < lines.length; i++) {
        const line = lines[i];
        if (!line.includes('Claude') && !line.includes('claude') && !line.includes('CLAUDE')) continue;

        // Classify the reference
        let classification: string;
        if (/`[^`]*claude[^`]*`/i.test(line)) {
          classification = 'model-id (skip)';
        } else if (/CLAUDE\.md/.test(line)) {
          classification = 'filename (replace → AGENTS.md)';
        } else if (/Claude Code/i.test(line)) {
          classification = 'runtime-name (replace → coding agent)';
        } else if (/(?:^|[.!?]\s+)Claude\s/m.test(line)) {
          classification = 'identity-subject (replace → You/agent)';
        } else {
          classification = 'identity (replace)';
        }

        matches.push({ line: i + 1, text: line.trim(), classification });
      }

      if (matches.length > 0) {
        process.stdout.write(`${relativePath}: ${matches.length} reference(s)\n`);
        for (const m of matches) {
          const truncated = m.text.length > 100 ? m.text.slice(0, 100) + '...' : m.text;
          process.stdout.write(`  L${m.line} [${m.classification}]: ${truncated}\n`);
        }
        process.stdout.write('\n');
        totalMatches += matches.length;
      }
    } catch {
      process.stdout.write(`${relativePath}: error reading\n`);
    }
  }

  process.stdout.write(`\nTotal: ${totalMatches} Claude reference(s) across ${allFiles.length} file(s)\n`);
}
```
