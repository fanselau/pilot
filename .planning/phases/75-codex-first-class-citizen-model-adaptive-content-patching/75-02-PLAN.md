**2. prepareProjectForSpawn:**
```typescript
import { patchAgentFrontmatter, resolveAllAgentModels } from './models.js';
import { installSkillsForJob } from './skills.js';
import { patchContentForModel, restoreContent, getModelFamily } from './prompt-adapter.js';
import type { ModelProfile, ModelEntry } from './types.js';

export async function prepareProjectForSpawn(
  projectDir: string,
  job: { categories: string[] | null; modelProfile: string | null; providerMode: string },
  models: Record<string, ModelEntry>,
  providerMode: string,
): Promise<PrepState> {
  const state: PrepState = {
    originals: new Map(),
    injectedSkills: [],
    projectDir,
    patchedAgents: false,
  };

  // Step 1: Patch agent frontmatter (model/variant)
  if (job.modelProfile) {
    const summary = patchAgentFrontmatter(projectDir, models);
    state.patchedAgents = true;
    const fallbackAgents = Array.isArray(summary?.fallback) ? summary.fallback : [];
    const skippedAgents = Array.isArray(summary?.skipped) ? summary.skipped : [];
    if (fallbackAgents.length > 0) {
      process.stderr.write(
        `[spawn-prep] Agent model fallback to inherit (${fallbackAgents.length}): ${fallbackAgents.join(', ')}\n`,
      );
    }
    if (skippedAgents.length > 0) {
      process.stderr.write(
        `[spawn-prep] Warning: skipped agent model patch (${skippedAgents.length}): ${skippedAgents.join(', ')}\n`,
      );
    }
  }

  // Step 2: Install skills via JIT npx (must happen BEFORE content patching — skill files also need adaptation)
  try {
    const installed = await installSkillsForJob(job.categories, projectDir);
    state.injectedSkills = installed;
    if (installed.length > 0) {
      process.stderr.write(
        `[spawn-prep] Installed ${installed.length} skill(s): ${installed.join(', ')}\n`,
      );
    }
  } catch (err) {
    process.stderr.write(
      `[spawn-prep] Warning: skill installation failed: ${String(err)}\n`,
    );
  }

  // Step 3: Patch content for target model family (identity/tool/path neutralization)
  const contentPatch = patchContentForModel(projectDir, providerMode, models);
  // Merge content originals into state
  for (const [filePath, original] of contentPatch.originals) {
    state.originals.set(filePath, original);
  }

  return state;
}
```

**3. restoreProjectAfterSpawn:**
```typescript
import { existsSync, rmSync } from 'node:fs';
import path from 'node:path';
import { restoreContent } from './prompt-adapter.js';

export function restoreProjectAfterSpawn(state: PrepState): void {
  // Restore content patches (originals back to disk)
  if (state.originals.size > 0) {
    restoreContent({ originals: state.originals, projectDir: state.projectDir });
  }

  // Clean up JIT-installed skills
  if (state.injectedSkills.length > 0) {
    const skillDir = path.join(state.projectDir, '.opencode', 'skill');
    if (existsSync(skillDir)) {
      try {
        rmSync(skillDir, { recursive: true, force: true });
      } catch {
        process.stderr.write(`[spawn-prep] Warning: failed to cleanup skills at ${skillDir}\n`);
      }
    }
  }

  // Note: agent frontmatter patches are NOT restored — they're per-project configuration
  // that stays until next patchAgentFrontmatter() call. This matches existing behavior.
}
```
