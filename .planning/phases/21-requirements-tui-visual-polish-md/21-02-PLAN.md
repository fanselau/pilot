**app.tsx — Add detail view handlers.** After the dashboard `if (state.view() === 'dashboard')` block, add:
```typescript
// Detail view navigation
if (state.view() === 'detail') {
  // j / down — scroll down
  if (key.name === 'j' || key.name === 'down') {
    state.setDetailScrollOffset(o => o + 1);
    state.setFollowLog(false);
    return;
  }
  // k / up — scroll up
  if (key.name === 'k' || key.name === 'up') {
    state.setDetailScrollOffset(o => Math.max(o - 1, 0));
    state.setFollowLog(false);
    return;
  }
  // f — toggle follow mode
  if (key.sequence === 'f') {
    state.setFollowLog(f => !f);
    return;
  }
  // g — jump to top
  if (key.sequence === 'g' && !key.shift) {
    state.setDetailScrollOffset(0);
    state.setFollowLog(false);
    return;
  }
  // G — jump to bottom (enable follow)
  if (key.sequence === 'G' || (key.sequence === 'g' && key.shift)) {
    state.setFollowLog(true);
    return;
  }
  // x — cancel this job (with confirmation if running)
  if (key.sequence === 'x') {
    const job = state.selectedJob();
    if (job) {
      if (job.status === 'running') {
        state.setConfirmAction({ type: 'cancel', jobId: job.id });
      }
    }
    return;
  }
  // r — retry this job
  if (key.sequence === 'r') {
    const job = state.selectedJob();
    if (job && job.status === 'failed') {
      retry(job.id);
    }
    return;
  }
}
```
