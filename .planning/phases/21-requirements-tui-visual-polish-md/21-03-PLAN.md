3. Daemon status indicator — a simple "daemon: on/off" (check if runner PID file exists, or simpler: just check if there are running jobs)

Updated StatusBar:
```typescript
export function StatusBar(props: {
  queue: Job[];
  running: Job[];
  completed: Job[];
  totalTokens?: number;
}) {
  const [time, setTime] = createSignal(new Date().toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit' }));

  let timer: ReturnType<typeof setInterval> | null = null;
  onMount(() => {
    timer = setInterval(() => {
      setTime(new Date().toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit' }));
    }, 10_000); // every 10s is fine for clock
  });
  onCleanup(() => { if (timer) clearInterval(timer); });

  const daemonIndicator = () => props.running.length > 0 ? '● daemon' : '○ daemon';
  const daemonColor = () => props.running.length > 0 ? statusColors.done : theme.muted;

  const tokenStr = () => {
    const t = props.totalTokens ?? 0;
    if (t === 0) return '';
    return `◆ ${formatTokens(t)} tok  `;
  };

  const summary = () =>
    `${tokenStr()}${props.queue.length} queued  ${props.running.length} running  ${props.completed.length} done  ${time()}`;

  return (
    <box flexDirection="row" height={1}>
      <text content=" PILOT " fg={statusColors.running} attributes={1} />
      <text content={` ${daemonIndicator()}`} fg={daemonColor()} />
      <box flexGrow={1} />
      <text content={`${summary()} `} fg={theme.muted} />
    </box>
  );
}
```

Import `formatTokens` from `'../utils/format.js'` (created in Plan 01), `createSignal`, `onMount`, `onCleanup` from 'solid-js'.

Update app.tsx to pass `totalTokens` prop to StatusBar. Compute it by summing all sessionTokens values:
```typescript
const totalTokens = () => {
  let sum = 0;
  for (const [, t] of state.sessionTokens()) {
    sum += t.input + t.output;
  }
  // Also add completed tokens
  for (const [, t] of state.completedTokens()) {
    sum += t;
  }
  return sum;
};
```
